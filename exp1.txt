-- ── 1. TABLE CREATION ──────────────────────────────────

CREATE TABLE Branch (
    branch_id INT PRIMARY KEY,
    branch_name VARCHAR2(50) NOT NULL,
    city VARCHAR2(30),
    address VARCHAR2(100)
);

CREATE TABLE Customer (
    customer_id INT PRIMARY KEY,
    name VARCHAR2(50) NOT NULL,
    address VARCHAR2(100),
    phone VARCHAR2(15),
    email VARCHAR2(50)
);

CREATE TABLE Account (
    account_no INT PRIMARY KEY,
    account_type VARCHAR2(10) NOT NULL,
    balance NUMBER(12,2) DEFAULT 0.00,
    open_date DATE,
    customer_id INT,
    branch_id INT,
    FOREIGN KEY (customer_id) REFERENCES Customer(customer_id),
    FOREIGN KEY (branch_id) REFERENCES Branch(branch_id),
    CONSTRAINT chk_account_type CHECK (account_type IN ('Savings', 'Current'))
);

CREATE TABLE Transactions (
    txn_id INT PRIMARY KEY,
    txn_date DATE,
    txn_type VARCHAR2(15),
    amount NUMBER(10,2),
    account_no INT,
    FOREIGN KEY (account_no) REFERENCES Account(account_no),
    CONSTRAINT chk_txn_type CHECK (txn_type IN ('Deposit', 'Withdrawal'))
);

CREATE TABLE Loan (
    loan_id INT PRIMARY KEY,
    loan_type VARCHAR2(30),
    loan_amount NUMBER(12,2),
    interest_rate NUMBER(5,2),
    branch_id INT,
    FOREIGN KEY (branch_id) REFERENCES Branch(branch_id)
);

CREATE TABLE Customer_Loan (
    customer_id INT,
    loan_id INT,
    apply_date DATE,
    PRIMARY KEY (customer_id, loan_id),
    FOREIGN KEY (customer_id) REFERENCES Customer(customer_id),
    FOREIGN KEY (loan_id) REFERENCES Loan(loan_id)
);
